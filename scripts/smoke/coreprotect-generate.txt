# Makes a real CoreProtect database for coreprotect-import.txt. Run it on a server that has CoreProtect
# and not Hawkeye, on a Minecraft version that CoreProtect build accepts, with a flat overworld
# (level-type=minecraft:flat). Then copy plugins/CoreProtect (config.yml and database.db) to the Hawkeye
# server. CoreProtect logs under #tnt, #hopper and #kill: stone and two stairs blown up, a named and
# enchanted sword, sticks and a coloured potion moved through a hopper, a zombie and a cow killed.
forceload add 0 0
fill 0 -60 0 6 -57 6 minecraft:stone
setblock 2 -56 2 minecraft:oak_stairs[facing=east,half=top]
setblock 4 -56 4 minecraft:oak_stairs
summon minecraft:tnt 3 -56 3 {fuse:1}
execute if block 3 -57 3 minecraft:stone => Test failed
setblock 20 -60 20 minecraft:chest
setblock 20 -59 20 minecraft:hopper[facing=down]
setblock 20 -58 20 minecraft:chest{Items:[{Slot:0b,id:"minecraft:diamond_sword",count:1,components:{"minecraft:custom_name":"Excalibur","minecraft:enchantments":{"minecraft:sharpness":5}}},{Slot:1b,id:"minecraft:stick",count:3},{Slot:2b,id:"minecraft:potion",count:1,components:{"minecraft:potion_contents":{potion:"minecraft:swiftness",custom_color:16711680}}}]}
summon minecraft:zombie 10 -60 10 {NoAI:1b}
summon minecraft:cow 12 -60 12 {NoAI:1b}
say waiting for the hopper
say waiting for the hopper
say waiting for the hopper
kill @e[type=minecraft:zombie]
kill @e[type=minecraft:cow]
say waiting for CoreProtect to write
say waiting for CoreProtect to write
say waiting for CoreProtect to write
